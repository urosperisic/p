FROM + JOIN → uzmi podatke
WHERE → izbaci pojedinačne redove koji ne odgovaraju
GROUP BY → napravi grupe
COUNT(*) → prebroj redove u svakoj grupi
HAVING → izbaci grupe koje imaju manje od 5
SELECT → odredi šta će se prikazati

---------- annotate vs aggregate

Book.objects.aggregate(Count('id'))
SELECT COUNT(id) AS id__count FROM book;

Book.objects.aggregate(total=Count('id'))
SELECT COUNT(id) AS total FROM book;

Book.objects.aggregate(avg_year=Avg('published_year'))
SELECT AVG(published_year) AS avg_year FROM book;

Author.objects.aggregate(oldest=Min('birth_year'), newest=Max('birth_year'))
SELECT MIN(birth_year) AS oldest, MAX(birth_year) AS newest FROM author

authors = Author.objects.annotate(book_count=Count('books'))
for a in authors:
    print(a.name, a.book_count)

SELECT
    author.id,
    author.name,
    author.birth_year,
    COUNT(book.id) AS book_count
FROM author
LEFT OUTER JOIN book
    ON author.id = book.author_id
GROUP BY author.id;

from django.db.models import Q

authors = Author.objects.annotate(
    available_count=Count('books', filter=Q(books__is_available=True))
)
for a in authors:
    print(a.name, a.available_count)

SELECT
    author.id,
    author.name,
    author.birth_year,
    COUNT(book.id) FILTER (WHERE book.is_available) AS available_count
FROM author
LEFT OUTER JOIN book
    ON author.id = book.author_id
GROUP BY author.id;

Author.objects.annotate(book_count=Count('books')).filter(book_count__gte=3)

SELECT
    author.id,
    author.name,
    author.birth_year,
    COUNT(book.id) AS book_count
FROM author
LEFT OUTER JOIN book
    ON author.id = book.author_id
GROUP BY author.id
HAVING COUNT(book.id) >= 3
LIMIT 21;
